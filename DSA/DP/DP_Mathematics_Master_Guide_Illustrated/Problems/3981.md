## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int interleaveCharacters(string word1, string word2, string target) {
        const int MOD = 1000000007;

        int n = word1.size();
        int m = word2.size();

        vector<vector<long long> > dp(n + 1,
                                      vector<long long>(m + 1, 0));

        dp[0][0] = 1;

        for (char c : target) {

            vector<vector<long long> > col(n + 1,
                                           vector<long long>(m + 1, 0));

            vector<vector<long long> > row(n + 1,
                                           vector<long long>(m + 1, 0));

            // Column prefix
            for (int j = 0; j <= m; ++j) {
                for (int i = 1; i <= n; ++i) {
                    col[i][j] =
                        (col[i - 1][j] + dp[i - 1][j]) % MOD;
                }
            }

            // Row prefix
            for (int i = 0; i <= n; ++i) {
                for (int j = 1; j <= m; ++j) {
                    row[i][j] =
                        (row[i][j - 1] + dp[i][j - 1]) % MOD;
                }
            }

            vector<vector<long long> > ndp(n + 1,
                                           vector<long long>(m + 1, 0));

            for (int i = 0; i <= n; ++i) {
                for (int j = 0; j <= m; ++j) {

                    // Take current character from word1
                    if (i > 0 && word1[i - 1] == c)
                        ndp[i][j] += col[i][j];

                    // Take current character from word2
                    if (j > 0 && word2[j - 1] == c)
                        ndp[i][j] += row[i][j];

                    ndp[i][j] %= MOD;
                }
            }

            dp.swap(ndp);
        }

        long long ans = 0;

        // Must use at least one character from both strings
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= m; ++j) {
                ans = (ans + dp[i][j]) % MOD;
            }
        }

        return ans;
    }
};
```
