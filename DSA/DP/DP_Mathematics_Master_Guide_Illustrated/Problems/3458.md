## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    bool maxSubstringLength(string s, int k) {
        if (k == 0) return true;

        int n = s.size();
        int first[26], last[26];

        fill(first, first + 26, n);
        fill(last, last + 26, -1);

        for (int i = 0; i < n; ++i) {
            int c = s[i] - 'a';
            first[c] = min(first[c], i);
            last[c] = i;
        }

        vector<pair<int, int>> intervals;

        for (int c = 0; c < 26; ++c) {
            if (last[c] == -1) continue;

            int l = first[c], r = last[c];
            bool valid = true;

            for (int i = l; i <= r; ++i) {
                int x = s[i] - 'a';

                if (first[x] < l) {
                    valid = false;
                    break;
                }

                r = max(r, last[x]);
            }

            if (valid && (l != 0 || r != n - 1))
                intervals.push_back({r, l});
        }

        sort(intervals.begin(), intervals.end());

        int end = -1;

        for (auto &p : intervals) {
            if (p.second > end) {
                end = p.first;
                if (--k == 0) return true;
            }
        }

        return false;
    }
};
```
