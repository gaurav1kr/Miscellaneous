## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int minOperations(string s1, string s2) {
        const int INF = 1e9;
        int n = s1.size();

        int dp0 = 0;
        int dp1 = INF;

        for (int i = 0; i < n; ++i) {
            int next0 = INF;
            int next1 = INF;

            // Current bit is original s1[i]
            if (dp0 < INF) {
                int cur = s1[i] - '0';
                int tar = s2[i] - '0';

                // Don't use pair operation
                if (cur == tar)
                    next0 = min(next0, dp0);
                else if (cur == 0 && tar == 1)
                    next0 = min(next0, dp0 + 1);

                // Use 11 -> 00 on i, i+1
                if (i + 1 < n) {
                    int cost = 1;

                    if (cur == 0) ++cost;
                    if (s1[i + 1] == '0') ++cost;
                    if (tar == 1) ++cost;

                    next1 = min(next1, dp0 + cost);
                }
            }

            // Current bit was already changed to 0
            // by pair operation from previous position
            if (dp1 < INF) {
                int cur = 0;
                int tar = s2[i] - '0';

                // Don't use pair operation
                if (cur == tar)
                    next0 = min(next0, dp1);
                else // 0 -> 1
                    next0 = min(next0, dp1 + 1);

                // Use pair operation on i, i+1
                if (i + 1 < n) {
                    int cost = 1;

                    // cur is 0, so first make it 1
                    ++cost;

                    if (s1[i + 1] == '0')
                        ++cost;

                    if (tar == 1)
                        ++cost;

                    next1 = min(next1, dp1 + cost);
                }
            }

            dp0 = next0;
            dp1 = next1;
        }

        return dp0 == INF ? -1 : dp0;
    }
};
```
