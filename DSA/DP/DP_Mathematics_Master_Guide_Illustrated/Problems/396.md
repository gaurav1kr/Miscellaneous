## Your original Accepted C++ submission (unaltered)

```cpp
#include <vector>

using namespace std;

class Solution {
public:
    int maxRotateFunction(vector<int>& nums) {
        int n = nums.size();
        long sum = 0, f = 0;
        for (int i = 0; i < n; ++i) {
            sum += nums[i];
            f += i * nums[i];
        }

        long res = f;
        for (int i = n - 1; i > 0; --i) {
            f += sum - (long)n * nums[i];
            res = max(res, f);
        }

        return res;
    }
};
```
