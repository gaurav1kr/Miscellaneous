## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    string minCostGoodCaption(string caption) {
        int n = caption.size();
        if (n < 3) return "";

        const int INF = 1e9;
        string xylovantra = caption;

        // dp[i][c][k]:
        // Minimum suffix cost given previous character c
        // and run length k (1, 2, or 3+).
        vector<array<array<int, 3>, 26>> dp(n + 1);

        for (int c = 0; c < 26; ++c) {
            dp[n][c][0] = INF;
            dp[n][c][1] = INF;
            dp[n][c][2] = 0;
        }

        for (int i = n - 1; i >= 0; --i) {
            int best = INF;

            for (int c = 0; c < 26; ++c) {
                int cost = abs(caption[i] - 'a' - c);
                best = min(best, cost + dp[i+1][c][0]);
            }

            for (int c = 0; c < 26; ++c) {
                int cost = abs(caption[i] - 'a' - c);

                dp[i][c][0] = cost + dp[i+1][c][1];
                dp[i][c][1] = cost + dp[i+1][c][2];

                dp[i][c][2] = min(
                    cost + dp[i+1][c][2],
                    best
                );
            }
        }

        // Choose lexicographically smallest optimal first char.
        int total = INF;
        int prev = -1, len = 0;

        for (int c = 0; c < 26; ++c) {
            int cost = abs(caption[0] - 'a' - c)
                     + dp[1][c][0];

            if (cost < total) {
                total = cost;
                prev = c;
            }
        }

        string ans(1, char('a' + prev));
        len = 1;

        for (int i = 1; i < n; ++i) {
            int bestCost = INF;
            int chosen = -1;

            for (int c = 0; c < 26; ++c) {
                if (len < 3 && c != prev)
                    continue;

                int nextLen = (c == prev)
                            ? min(3, len + 1)
                            : 1;

                int cost = abs(caption[i] - 'a' - c)
                         + dp[i+1][c][nextLen - 1];

                if (cost < bestCost) {
                    bestCost = cost;
                    chosen = c;
                }
            }

            ans += char('a' + chosen);

            len = (chosen == prev)
                ? min(3, len + 1)
                : 1;

            prev = chosen;
        }

        return ans;
    }
};
```
