## Your original Accepted C++ submission (unaltered)

```cpp
#include <vector>
#include <string>

using namespace std;

class Solution {
public:
    int numDupDigitsAtMostN(int n) {
        string s = to_string(n + 1);
        int len = s.size(), res = 0;

        for (int i = 1; i < len; ++i)
            res += 9 * perm(9, i - 1);

        vector<bool> used(10);
        for (int i = 0; i < len; ++i) {
            int d = s[i] - '0';
            for (int j = i == 0 ? 1 : 0; j < d; ++j)
                if (!used[j])
                    res += perm(9 - i, len - i - 1);
            if (used[d]) break;
            used[d] = true;
        }
        return n - res;
    }

    int perm(int m, int n) {
        int res = 1;
        while (n--) res *= m--;
        return res;
    }
};
```
