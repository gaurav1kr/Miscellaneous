## Understand the mathematics first

Carry XOR value as an extra state dimension.

### Model / useful recurrence

`W(i,j,x)=W(i-1,j,x XOR grid(i,j))+W(i,j-1,x XOR grid(i,j))`

**Why this helps:** XOR is invertible: previous XOR = desired XOR XOR current value.

## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int countPathsWithXorValue(vector<vector<int>>& grid,
                               int k) {
        const int MOD = 1000000007;
        int m = grid.size(), n = grid[0].size();

        // dp[j][x] = number of paths reaching
        // current row's column j with XOR x
        vector<array<int, 16>> dp(n);

        dp[0][grid[0][0]] = 1;

        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                if (i == 0 && j == 0)
                    continue;

                array<int, 16> curr = {};

                for (int x = 0; x < 16; ++x) {
                    int prev = x ^ grid[i][j];

                    // From top
                    if (i > 0) {
                        curr[x] = dp[j][prev];
                    }

                    // From left
                    if (j > 0) {
                        curr[x] = (curr[x] +
                                   dp[j-1][prev]) % MOD;
                    }
                }

                dp[j] = curr;
            }
        }

        return dp[n-1][k];
    }
};
```
