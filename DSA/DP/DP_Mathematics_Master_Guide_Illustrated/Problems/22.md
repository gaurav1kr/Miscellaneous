Balanced parentheses are generated by constrained prefixes.

### Model / useful recurrence

`open<=n and close<=open`

## Your original Accepted C++ submission (unaltered)

```cpp
class Solution
{
    vector<string> str ;
public:
    void brackets(string output, int open, int close, int pairs , int n)
    {
        if(open == pairs && close == pairs && (output.length() == 2*n))
            str.push_back(output);
        else
        {
            if(open<pairs)
                    brackets(output+"(",open+1,close,pairs,n);
            if(close<open)
                    brackets(output+")",open,close+1,pairs,n);
        }
    }
    vector<string> generateParenthesis(int n) 
    {
         for(int i=1;i<=n;i++)
         {
            brackets("",0,0,i,n);
         }
         return str ;
    }
};
```
