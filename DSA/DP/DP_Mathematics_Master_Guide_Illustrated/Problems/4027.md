```cpp
class Solution {
public:
    long long elevatorRequests(int n, int start,
                               vector<vector<int>>& requests) {
        
        int m = requests.size();
        int totalMasks = 1 << m;

        const long long INF = (1LL << 60);

        // dp[mask][i] =
        // minimum time after fulfilling all requests in mask,
        // with request i fulfilled last.
        vector<vector<long long>> dp(
            totalMasks,
            vector<long long>(m, INF)
        );

        // ----------------------------------
        // Initialize first request
        // ----------------------------------
        for (int i = 0; i < m; ++i) {

            long long travel =
                llabs((long long)start - requests[i][1]);

            dp[1 << i][i] =
                max(travel,
                    (long long)requests[i][0]);
        }

        // ----------------------------------
        // Bitmask DP
        // ----------------------------------
        for (int mask = 1; mask < totalMasks; ++mask) {

            for (int i = 0; i < m; ++i) {

                // i must be the last fulfilled request
                if (!(mask & (1 << i)))
                    continue;

                if (dp[mask][i] == INF)
                    continue;

                // Choose next request j
                for (int j = 0; j < m; ++j) {

                    if (mask & (1 << j))
                        continue;

                    long long distance =
                        llabs((long long)requests[i][1]
                              - requests[j][1]);

                    long long reachTime =
                        dp[mask][i] + distance;

                    // If we arrive before request j appears,
                    // wait until its arrival time.
                    long long newTime =
                        max(reachTime,
                            (long long)requests[j][0]);

                    int newMask =
                        mask | (1 << j);

                    dp[newMask][j] =
                        min(dp[newMask][j],
                            newTime);
                }
            }
        }

        // ----------------------------------
        // Answer
        // ----------------------------------
        int fullMask = totalMasks - 1;

        long long answer = INF;

        for (int i = 0; i < m; ++i) {
            answer = min(answer,
                         dp[fullMask][i]);
        }

        return answer;
    }
};
```
