## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int minMaxWaitingTime(vector<int>& d, vector<int>& fuel) {
        vector<int> telmorvian = d;

        // state = {fuel0, fuel1, busy0, busy1, maxWait}
        vector<array<int,5> > cur;
        cur.push_back({fuel[0], fuel[1], 0, 0, 0});

        int ans = -1;

        for (int x : d) {
            map<array<int,4>, int> best;

            for (auto s : cur) {
                for (int p = 0; p < 2; ++p) {
                    int f0 = s[0], f1 = s[1];
                    int b0 = s[2], b1 = s[3];
                    int w  = s[4];

                    int &f = (p == 0 ? f0 : f1);
                    int &b = (p == 0 ? b0 : b1);

                    if (f < x) continue;

                    int wait = b;
                    f -= x;

                    if (p == 0) {
                        b0 = x;
                        b1 = max(0, b1 - wait);
                    } else {
                        b1 = x;
                        b0 = max(0, b0 - wait);
                    }

                    array<int,4> key = {f0, f1, b0, b1};
                    int nw = max(w, wait);

                    if (!best.count(key) || nw < best[key])
                        best[key] = nw;
                }
            }

            if (best.empty()) break;

            cur.clear();
            ans = INT_MAX;

            for (auto &e : best) {
                cur.push_back({
                    e.first[0], e.first[1],
                    e.first[2], e.first[3],
                    e.second
                });

                ans = min(ans, e.second);
            }
        }

        return ans;
    }
};
```
