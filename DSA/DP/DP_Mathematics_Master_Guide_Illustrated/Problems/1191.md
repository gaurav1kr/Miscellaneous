## Your original Accepted C++ submission (unaltered)

```cpp
#include <vector>
#include <algorithm>

using namespace std;

class Solution {
public:
    int kConcatenationMaxSum(vector<int>& arr, int k) {
        const int MOD = 1e9 + 7;
        long max_kadane = kadane(arr);
        if (k == 1) return max_kadane % MOD;

        long total = accumulate(arr.begin(), arr.end(), 0L);
        long double_kadane = kadane(concat(arr, arr));

        return max({max_kadane, double_kadane, total > 0 ? double_kadane + (k - 2) * total : double_kadane}) % MOD;
    }

    long kadane(const vector<int>& arr) {
        long max_sum = 0, cur = 0;
        for (int x : arr) {
            cur = max((long)x, cur + x);
            max_sum = max(max_sum, cur);
        }
        return max_sum;
    }

    vector<int> concat(const vector<int>& a, const vector<int>& b) {
        vector<int> res = a;
        res.insert(res.end(), b.begin(), b.end());
        return res;
    }
};
```
