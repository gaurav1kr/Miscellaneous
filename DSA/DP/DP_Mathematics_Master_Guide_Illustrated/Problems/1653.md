## Your original Accepted C++ submission (unaltered)

```cpp
#include <string>
#include <algorithm>

using namespace std;

class Solution {
public:
    int minimumDeletions(string s) {
        int rightA = count(s.begin(), s.end(), 'a');
        int res = rightA, leftB = 0;
        for (char c : s) {
            if (c == 'a') rightA--;
            else leftB++;
            res = min(res, leftB + rightA);
        }
        return res;
    }
};
```
