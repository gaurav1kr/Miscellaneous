## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int nthSuperUglyNumber(int n, vector<int>& primes) {
        int k = primes.size();

        vector<int> dp(n);
        vector<int> idx(k, 0);
        vector<long long> next(k);

        dp[0] = 1;

        for (int j = 0; j < k; ++j)
            next[j] = primes[j];

        for (int i = 1; i < n; ++i) {

            // Find next smallest ugly number
            long long mn = next[0];

            for (int j = 1; j < k; ++j)
                mn = min(mn, next[j]);

            dp[i] = (int)mn;

            // Move ALL pointers producing mn
            for (int j = 0; j < k; ++j) {
                if (next[j] == mn) {
                    ++idx[j];
                    next[j] = 1LL * primes[j] * dp[idx[j]];
                }
            }
        }

        return dp[n - 1];
    }
};
```
