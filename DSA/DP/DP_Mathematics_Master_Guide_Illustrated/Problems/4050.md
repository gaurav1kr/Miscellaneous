## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int minDays(int n) {
        const int INF = 1e9;

        vector<int> dp(n + 1, INF);
        dp[0] = 0;

        // Generate all possible streak lengths.
        for (int k = 1; ; ++k) {
            int points = k * (k + 1) / 2;

            if (points > n)
                break;

            int cost = k + 1;

            // Unbounded knapsack:
            // same streak length can be used multiple times.
            for (int score = points; score <= n; ++score) {
                dp[score] = min(
                    dp[score],
                    dp[score - points] + cost
                );
            }
        }

        // Last streak doesn't need a skip after it.
        return dp[n] - 1;
    }
};
```
