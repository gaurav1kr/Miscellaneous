## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
private:
    int m, n;
    vector<vector<int> > prefix;

    // Sum of rectangle:
    // [r1, c1] -> [r2, c2], inclusive
    int getSum(int r1, int c1, int r2, int c2) {
        return prefix[r2 + 1][c2 + 1]
             - prefix[r1][c2 + 1]
             - prefix[r2 + 1][c1]
             + prefix[r1][c1];
    }

    bool possible(int k) {
        int minRow = m;
        int maxRow = -1;

        int minCol = n;
        int maxCol = -1;

        int count = 0;

        for (int r = 0; r + k <= m; ++r) {
            for (int c = 0; c + k <= n; ++c) {

                // Check if entire k x k square is 1.
                if (getSum(r, c,
                           r + k - 1,
                           c + k - 1) == k * k) {

                    ++count;

                    minRow = min(minRow, r);
                    maxRow = max(maxRow, r);

                    minCol = min(minCol, c);
                    maxCol = max(maxCol, c);
                }
            }
        }

        if (count < 2)
            return false;

        // Two squares separated vertically
        if (maxRow - minRow >= k)
            return true;

        // Two squares separated horizontally
        if (maxCol - minCol >= k)
            return true;

        return false;
    }

public:
    int maxArea(vector<vector<int> >& mat) {
        m = mat.size();
        n = mat[0].size();

        // --------------------------------
        // Build 2D prefix sum
        // --------------------------------
        prefix.assign(m + 1,
                      vector<int>(n + 1, 0));

        for (int r = 0; r < m; ++r) {
            for (int c = 0; c < n; ++c) {

                prefix[r + 1][c + 1] =
                    mat[r][c]
                    + prefix[r][c + 1]
                    + prefix[r + 1][c]
                    - prefix[r][c];
            }
        }

        // --------------------------------
        // Binary search side length
        // --------------------------------
        int low = 1;
        int high = min(m, n);

        int best = 0;

        while (low <= high) {

            int mid = low + (high - low) / 2;

            if (possible(mid)) {
                best = mid;
                low = mid + 1;
            }
            else {
                high = mid - 1;
            }
        }

        return best * best;
    }
};
```
