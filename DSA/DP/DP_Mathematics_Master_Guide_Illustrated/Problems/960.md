## Your original Accepted C++ submission (unaltered)

```cpp
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

class Solution {
public:
    int minDeletionSize(vector<string>& strs) {
        int n = strs.size(), m = strs[0].size();
        vector<int> dp(m, 1);
        int res = m - 1;

        for (int j = 1; j < m; ++j)
            for (int i = 0; i < j; ++i) {
                bool valid = true;
                for (int k = 0; k < n; ++k)
                    if (strs[k][i] > strs[k][j]) {
                        valid = false;
                        break;
                    }
                if (valid) dp[j] = max(dp[j], dp[i] + 1);
            }

        return m - *max_element(dp.begin(), dp.end());
    }
};
```
