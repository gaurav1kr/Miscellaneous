## Understand the mathematics first

Carry remaining neutralizations as an extra state.

### Model / useful recurrence

`F(i,j,k)=max(fromTopOrLeft+value, fromTopOrLeftWithOneMoreNeutralization)`

**Why this helps:** A negative cell may be neutralized at most twice.

## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int maximumAmount(vector<vector<int>>& coins) {
        int m = coins.size(), n = coins[0].size();
        const int NEG = -1000000000;

        // dp[j][k]: max coins at column j
        // with k neutralizations remaining
        vector<array<int, 3>> dp(n, {NEG, NEG, NEG});

        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                int x = coins[i][j];

                if (i == 0 && j == 0) {
                    dp[j] = {NEG, NEG, x};
                    if (x < 0)
                        dp[j][1] = 0;
                    continue;
                }

                array<int, 3> prev = {NEG, NEG, NEG};

                for (int k = 0; k < 3; ++k) {
                    // From top or left
                    prev[k] = dp[j][k];
                    if (j > 0)
                        prev[k] = max(prev[k], dp[j-1][k]);
                }

                for (int k = 0; k < 3; ++k) {
                    // Accept current cell value
                    dp[j][k] = prev[k] + x;

                    // Neutralize negative cell
                    if (x < 0 && k < 2)
                        dp[j][k] = max(dp[j][k],
                                       prev[k+1]);
                }
            }
        }

        return max({dp[n-1][0], dp[n-1][1],
                    dp[n-1][2]});
    }
};
```
