## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    long long maxValue(vector<int>& nums) {
        int n = nums.size();

        // Alternating prefix sum
        vector<long long> P(n + 1, 0);

        for (int i = 0; i < n; ++i) {
            if (i % 2 == 0)
                P[i + 1] = P[i] + nums[i];
            else
                P[i + 1] = P[i] - nums[i];
        }

        long long original = P[n];
        long long maxGain = 0;

        const long long NEG = LLONG_MIN / 4;

        // For same parity:
        // we need max P[l+1]
        long long maxSame[2] = {NEG, NEG};

        // For different parity:
        // we need max P[l]
        long long maxDiff[2] = {NEG, NEG};

        // Scan r from left to right.
        for (int r = 1; r < n; ++r) {

            // Before processing r, add l = r-1.
            // This guarantees l < r.
            int l = r - 1;

            maxSame[l % 2] =
                max(maxSame[l % 2], P[l + 1]);

            maxDiff[l % 2] =
                max(maxDiff[l % 2], P[l]);

            // ---------------------------------
            // Case 1: l and r same parity
            // gain = 2 * (P[l+1] - P[r+1])
            // ---------------------------------
            if (maxSame[r % 2] != NEG) {
                long long gain =
                    2LL * (maxSame[r % 2] - P[r + 1]);

                maxGain = max(maxGain, gain);
            }

            // ---------------------------------
            // Case 2: l and r different parity
            // gain = 2 * (P[l] - P[r+1])
            // ---------------------------------
            int opposite = 1 - (r % 2);

            if (maxDiff[opposite] != NEG) {
                long long gain =
                    2LL * (maxDiff[opposite] - P[r + 1]);

                maxGain = max(maxGain, gain);
            }
        }

        return original + maxGain;
    }
};
```
