## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int minCost(string source, string target,
                vector<vector<string>>& rules,
                vector<int>& costs) {

        int n = source.size(), m = rules.size();
        const long long INF = 1e18;

        vector<long long> dp(n + 1, INF);
        dp[0] = 0;

        for (int i = 0; i < n; ++i) {
            if (dp[i] == INF) continue;

            // No operation needed at this position
            if (source[i] == target[i])
                dp[i + 1] = min(dp[i + 1], dp[i]);

            // Try every rule starting at i
            for (int j = 0; j < m; ++j) {
                string &p = rules[j][0];
                string &r = rules[j][1];

                int len = p.size();
                if (i + len > n) continue;

                bool ok = true;
                int cost = costs[j];

                for (int k = 0; k < len; ++k) {

                    // Replacement must exactly match target
                    if (r[k] != target[i + k]) {
                        ok = false;
                        break;
                    }

                    // Pattern must match source
                    if (p[k] == '*')
                        ++cost;
                    else if (p[k] != source[i + k]) {
                        ok = false;
                        break;
                    }
                }

                if (ok)
                    dp[i + len] =
                        min(dp[i + len],
                            dp[i] + cost);
            }
        }

        return dp[n] == INF ? -1 : (int)dp[n];
    }
};
```
