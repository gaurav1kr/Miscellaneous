## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int minOperations(vector<int>& nums, int target) {
        const int INF = 1e9;

        // dp[s] = minimum operations to form sum s
        vector<int> dp(target + 1, INF);
        dp[0] = 0;

        for (int x : nums) {

            // Generate only reachable values.
            // best[v] = minimum cost to transform x -> v
            vector<pair<int, int> > options;

            // Temporarily keep best cost.
            unordered_map<int, int> best;

            int cur = x;
            int divOps = 0;

            // x -> x/2 -> x/4 -> ...
            while (cur > 0) {

                long long value = cur;
                int mulOps = 0;

                // Then multiply by 2 repeatedly.
                while (value <= target) {

                    int v = (int)value;
                    int cost = divOps + mulOps;

                    auto it = best.find(v);

                    if (it == best.end() || cost < it->second)
                        best[v] = cost;

                    value *= 2;
                    ++mulOps;
                }

                cur /= 2;
                ++divOps;
            }

            // Convert to compact list.
            for (auto &p : best) {
                options.push_back({p.first, p.second});
            }

            // 0/1 knapsack
            vector<int> next = dp;

            for (int s = 0; s <= target; ++s) {

                if (dp[s] == INF)
                    continue;

                for (int j = 0; j < (int)options.size(); ++j) {

                    int value = options[j].first;
                    int cost  = options[j].second;

                    if (s + value <= target) {
                        next[s + value] =
                            min(next[s + value],
                                dp[s] + cost);
                    }
                }
            }

            dp.swap(next);

            // Can't do better than 0 operations.
            if (dp[target] == 0)
                return 0;
        }

        return dp[target] == INF ? -1 : dp[target];
    }
};
```
