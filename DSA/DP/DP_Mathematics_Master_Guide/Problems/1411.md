## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int numOfWays(int n) {
        const int MOD = 1e9 + 7;
        long a = 6, b = 6;

        for (int i = 2; i <= n; ++i) {
            long aNew = (2 * a + 2 * b) % MOD;
            long bNew = (2 * a + 3 * b) % MOD;
            a = aNew;
            b = bNew;
        }

        return (a + b) % MOD;
    }
};
```
