## Understand the mathematics first

Choose at most four mutually non-overlapping intervals.

### Model / useful recurrence

`F(i,k)=max(F(i+1,k),weight_i+F(next(i),k-1))`

**Why this helps:** Keep original sorted indices to break ties lexicographically.

## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
    struct State {
        long long score = 0;
        vector<int> ids;
    };

public:
    vector<int> maximumWeight(vector<vector<int>>& intervals) {
        int n = intervals.size();

        vector<array<int, 4>> a(n);
        for (int i = 0; i < n; ++i) {
            a[i] = {intervals[i][0], intervals[i][1],
                    intervals[i][2], i};
        }

        // Required by problem statement
        auto vorellixan = intervals;

        sort(a.begin(), a.end());

        vector<int> starts(n);
        for (int i = 0; i < n; ++i)
            starts[i] = a[i][0];

        // Find next non-overlapping interval
        vector<int> nxt(n);
        for (int i = 0; i < n; ++i) {
            nxt[i] = upper_bound(starts.begin(),
                                 starts.end(), a[i][1])
                     - starts.begin();
        }

        // dp[i][k] = best result from i onward,
        // selecting at most k intervals
        vector<array<State, 5>> dp(n + 1);

        for (int i = n - 1; i >= 0; --i) {
            for (int k = 1; k <= 4; ++k) {
                // Option 1: Skip
                dp[i][k] = dp[i + 1][k];

                // Option 2: Take
                State take = dp[nxt[i]][k - 1];
                take.score += a[i][2];

                int id = a[i][3];
                take.ids.insert(
                    lower_bound(take.ids.begin(),
                                take.ids.end(), id), id);

                // Max score, then lexicographically smallest
                if (take.score > dp[i][k].score ||
                    (take.score == dp[i][k].score &&
                     take.ids < dp[i][k].ids)) {
                    dp[i][k] = take;
                }
            }
        }

        return dp[0][4].ids;
    }
};
```
