### Model / useful recurrence

`F(i,j)=F(i-1,j)+F(i,j-1)`

**Why this helps:** For minimum cost paths, replace + of paths with min of predecessor costs, then add the current cell cost. Obstacles contribute zero ways.

```cpp
#include <vector>
#include <algorithm>
#include <unordered_set>

using namespace std;

class Solution {
public:
    int orderOfLargestPlusSign(int n, vector<vector<int>>& mines) {
        vector<vector<int>> grid(n, vector<int>(n, n));
        unordered_set<int> banned;
        for (auto& m : mines) {
            banned.insert(m[0] * n + m[1]);
        }

        for (int i = 0; i < n; ++i) {
            int count = 0;
            for (int j = 0; j < n; ++j) {
                count = banned.count(i * n + j) ? 0 : count + 1;
                grid[i][j] = min(grid[i][j], count);
            }
            count = 0;
            for (int j = n - 1; j >= 0; --j) {
                count = banned.count(i * n + j) ? 0 : count + 1;
                grid[i][j] = min(grid[i][j], count);
            }
        }

        for (int j = 0; j < n; ++j) {
            int count = 0;
            for (int i = 0; i < n; ++i) {
                count = banned.count(i * n + j) ? 0 : count + 1;
                grid[i][j] = min(grid[i][j], count);
            }
            count = 0;
            for (int i = n - 1; i >= 0; --i) {
                count = banned.count(i * n + j) ? 0 : count + 1;
                grid[i][j] = min(grid[i][j], count);
            }
        }

        int res = 0;
        for (auto& row : grid) {
            for (int x : row)
                res = max(res, x);
        }
        return res;
    }
};
```
