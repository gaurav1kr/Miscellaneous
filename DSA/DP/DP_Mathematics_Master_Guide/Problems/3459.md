**Why this helps:** For minimum cost paths, replace + of paths with min of predecessor costs, then add the current cell cost. Obstacles contribute zero ways.

## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
    int n, m;
    int dr[4] = {1, 1, -1, -1};
    int dc[4] = {1, -1, -1, 1};
    vector<int> dp;

    int dfs(vector<vector<int>>& grid, int r, int c,
            int d, int turn) {
        int id = ((r * m + c) * 4 + d) * 2 + turn;

        if (dp[id] != -1)
            return dp[id];

        int next = grid[r][c] == 1 ? 2 : 2 - grid[r][c];
        int ans = 1;

        int x = r + dr[d], y = c + dc[d];

        if (x >= 0 && x < n && y >= 0 && y < m &&
            grid[x][y] == next)
            ans = max(ans, 1 + dfs(grid, x, y, d, turn));

        if (turn) {
            int nd = (d + 1) % 4;
            x = r + dr[nd];
            y = c + dc[nd];

            if (x >= 0 && x < n && y >= 0 && y < m &&
                grid[x][y] == next)
                ans = max(ans, 1 + dfs(grid, x, y, nd, 0));
        }

        return dp[id] = ans;
    }

public:
    int lenOfVDiagonal(vector<vector<int>>& grid) {
        n = grid.size();
        m = grid[0].size();

        dp.assign(n * m * 8, -1);
        int ans = 0;

        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                if (grid[i][j] != 1) continue;

                for (int d = 0; d < 4; ++d)
                    ans = max(ans, dfs(grid, i, j, d, 1));
            }
        }

        return ans;
    }
};
```
