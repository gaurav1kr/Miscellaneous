## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int maxFrequency(vector<int>& nums, int k) {
        int dp[51] = {0};
        int base = 0, best = 0;

        for (int num : nums) {
            if (num == k) {
                ++base;
                for (int v = 1; v <= 50; ++v)
                    dp[v] = max(0, dp[v] - 1);
            } else {
                dp[num]++;
                best = max(best, dp[num]);
            }
        }

        return base + best;
    }
};
```
