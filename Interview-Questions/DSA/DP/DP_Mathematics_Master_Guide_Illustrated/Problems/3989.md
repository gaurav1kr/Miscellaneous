## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int maxConsistentColumns(vector<vector<int>>& grid, int limit) {
          int m = grid.size(), n = grid[0].size();
        vector<int> dp(n, 1);
        int ans = 1;

        for (int j = 0; j < n; ++j) {
            for (int k = 0; k < j; ++k) {
                bool ok = true;

                for (int i = 0; i < m; ++i) {
                    if (abs(grid[i][j] - grid[i][k]) > limit) {
                        ok = false;
                        break;
                    }
                }

                if (ok)
                    dp[j] = max(dp[j], dp[k] + 1);
            }

            ans = max(ans, dp[j]);
        }

        return ans;
    }
};
```
