## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    long long maxSubarraySum(vector<int>& nums) {
        long long ans = LLONG_MIN;
        long long prefix = 0, minPrefix = 0;
        long long bestMin = 0;

        unordered_map<int, long long> removed;

        for (int x : nums) {
            prefix += x;

            // Best subarray ending here
            ans = max(ans, prefix - bestMin);

            if (x < 0) {
                // Minimum adjusted prefix after
                // removing occurrences of x
                removed[x] = min(removed[x], minPrefix) + x;

                bestMin = min(bestMin, removed[x]);
            }

            minPrefix = min(minPrefix, prefix);
            bestMin = min(bestMin, minPrefix);
        }

        return ans;
    }
};
```
