## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int findIntegers(int n) {
        vector<int> dp(31, 0);
        dp[0] = 1;
        dp[1] = 2;

        for (int i = 2; i < 31; ++i)
            dp[i] = dp[i - 1] + dp[i - 2];

        int res = 0, prevBit = 0;
        for (int i = 29; i >= 0; --i) {
            if ((n & (1 << i)) != 0) {
                res += dp[i];
                if (prevBit) return res;
                prevBit = 1;
            } else {
                prevBit = 0;
            }
        }

        return res + 1;
    }
};
```
