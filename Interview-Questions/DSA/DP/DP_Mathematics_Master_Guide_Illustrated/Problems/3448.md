## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    long long countSubstrings(string s) {
        int dp[10][10] = {};
        long long ans = 0;

        for (char ch : s) {
            int digit = ch - '0';
            int next[10][10] = {};

            for (int d = 1; d <= 9; ++d) {
                ++next[d][digit % d];

                for (int r = 0; r < d; ++r) {
                    next[d][(r * 10 + digit) % d] += dp[d][r];
                }
            }

            for (int d = 1; d <= 9; ++d)
                for (int r = 0; r < d; ++r)
                    dp[d][r] = next[d][r];

            if (digit != 0)
                ans += dp[digit][0];
        }

        return ans;
    }
};
```
