## Your original Accepted C++ submission (unaltered)

```cpp
#include <vector>
#include <algorithm>

using namespace std;

class Solution {
public:
    int minSideJumps(vector<int>& obs) {
        int n = obs.size() - 1;
        vector<int> dp = {1, 0, 1};
        for (int i = 1; i <= n; ++i) {
            for (int j = 0; j < 3; ++j) {
                if (obs[i] == j + 1) dp[j] = 1e9;
            }
            for (int j = 0; j < 3; ++j) {
                if (obs[i] != j + 1) {
                    dp[j] = min(dp[j], min(dp[(j+1)%3], dp[(j+2)%3]) + 1);
                }
            }
        }
        return min({dp[0], dp[1], dp[2]});
    }
};
```
