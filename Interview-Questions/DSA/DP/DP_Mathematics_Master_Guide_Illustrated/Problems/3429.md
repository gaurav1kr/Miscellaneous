## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    long long minCost(int n, vector<vector<int>>& cost) {
        const long long INF = 1LL << 60;
        long long dp[3][3], ndp[3][3];

        for (int a = 0; a < 3; ++a)
            for (int b = 0; b < 3; ++b)
                dp[a][b] = (a == b) ? INF :
                    (long long)cost[0][a] + cost[n-1][b];

        for (int i = 1; i < n / 2; ++i) {
            for (int a = 0; a < 3; ++a) {
                for (int b = 0; b < 3; ++b) {
                    ndp[a][b] = INF;
                    if (a == b) continue;

                    for (int x = 0; x < 3; ++x) {
                        for (int y = 0; y < 3; ++y) {
                            if (a == x || b == y) continue;

                            ndp[a][b] = min(ndp[a][b],
                                dp[x][y] + cost[i][a]
                                         + cost[n-1-i][b]);
                        }
                    }
                }
            }

            for (int a = 0; a < 3; ++a)
                for (int b = 0; b < 3; ++b)
                    dp[a][b] = ndp[a][b];
        }

        long long ans = INF;
        for (int a = 0; a < 3; ++a)
            for (int b = 0; b < 3; ++b)
                ans = min(ans, dp[a][b]);

        return ans;
    }
};
```
