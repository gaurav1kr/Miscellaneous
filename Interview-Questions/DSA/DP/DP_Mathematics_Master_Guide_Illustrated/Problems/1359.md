## Your original Accepted C++ submission (unaltered)

```cpp
class Solution {
public:
    int countOrders(int n) {
        const int MOD = 1e9 + 7;
        long res = 1;
        for (int i = 1; i <= n; ++i) {
            res = res * i % MOD;
            res = res * (2 * i - 1) % MOD;
        }
        return res;
    }
};
```
